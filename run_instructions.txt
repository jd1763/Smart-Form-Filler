# ----------------------------
# Setup & Run Instructions (Source of Truth)
# ----------------------------

# This repo supports TWO backend modes:
#   v1 (Flask)   = Local-only single “Local Profile” in backend/data (no accounts)
#   v2 (FastAPI) = Multi-user + JWT auth (accounts)
#
# The Chrome extension uses Option A:
#   background.js is the ONLY resolver for base URL + mode routing.
#   UI pages do NOT probe ports directly. You pick v1/v2 using the popup toggle.

# ----------------------------
# Quick Start (Local, no Docker)
# ----------------------------

# 1) Create a virtual environment with Python 3.12
python3 -m venv .venv

# 2) Activate it (every new shell)
source .venv/bin/activate

# 3) Install deps
python -m pip install --upgrade pip
pip install -r requirements.txt

# 4) (Optional) spaCy model (only if you see missing model errors)
python -m spacy download en_core_web_sm


# ----------------------------
# Run Backend — Choose v1 or v2
# ----------------------------

# --- v1 (Flask / Local Mode) ---
# - Local-only single profile in backend/data/profile.json
# - Resumes live in backend/data/resumes/
# - No login/logout
#
# Local port auto-selects from: 5000–5004
# Optional override: SFF_PORT=5000
python -m backend.api


# --- v2 (FastAPI / Cloud Mode) ---
# - Multi-user accounts + JWT
# - Login/registration enabled
#
# Local port auto-selects from: 6000–6004
# Optional override: SFF_PORT_V2=6000
python -m backend.fastapi_app


# ----------------------------
# Run BOTH backends at once (recommended for development)
# ----------------------------

# This starts v1 + v2 together and cleans up old processes on those ports first.
# Script location: script/run_local_backends.sh
#
# It:
#  - kills old servers on v1 pool 5000–5004
#  - kills old servers on v2 pool 6001–6004 (avoids 6000)
#  - starts python -m backend.api   (v1)
#  - starts python -m backend.fastapi_app (v2)
#  - Ctrl+C stops both
bash script/run_local_backends.sh


# ----------------------------
# Load the Chrome Extension
# ----------------------------

# 1) Open chrome://extensions
# 2) Enable "Developer mode"
# 3) Click "Load unpacked"
# 4) Select: smart-form-filler/extension

# Usage:
# - Open a job posting page
# - Open the extension popup
# - Select backend mode (v1 or v2) using the toggle:
#     - v1 = Local Profile stored on this computer
#     - v2 = Sign in / Register (JWT auth)
# - Click "Fill Form"


# ----------------------------
# Run with Docker (one command)
# ----------------------------
# Preferred: use the Makefile shortcuts
#   make up        # build + start prod-like (@ http://localhost:8000)
#   make dev       # start dev hot-reload (@ http://localhost:8000)
#   make down      # stop
#   make logs      # tail logs
#   make rebuild   # clean rebuild

# Raw docker compose (alternative if I don't want to use make):
# Build + start
docker compose up --build backend        # prod-like (Gunicorn @ 8000)
# or:
docker compose up --build backend-dev    # dev hot-reload (@ 8000)

# Stop
docker compose down


# ----------------------------
# Notes / Overrides
# ----------------------------
